\DeclareBibliographyCategory{skeleton}

\RequirePackage{newfile}

\newoutputstream{skeleton}
\openoutputfile{\jobname.skeleton.bib}{skeleton}

\NewDocumentCommand{\skeleton}{O{article} m m O{} O{}}{
    \addtostream{skeleton}{%
        @#1{#2,
            author = {#3},
            year = {#4},
            title = {#5},
        }
    }
    \addtocategory{skeleton}{#2}
}

\addbibresource{\jobname.skeleton.bib}
