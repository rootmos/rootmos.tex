\usepackage{marginnote}

\newcounter{TLP}

\AddToHook{para/begin}{%
    \addtocounter{TLP}{1}%
    \ifthenelse{\value{TLP}=1}{\marginnote{Foo?}}{}%
}
\AddToHook{para/after}{%
    \addtocounter{TLP}{-1}%
}
